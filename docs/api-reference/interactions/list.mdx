`listInteractions` · Interaction

按真实 kind、state 与归属显示详情；没有题目或选项时不构造默认选择。

## TypeScript 参数与结果结构

以下结构直接摘自同版本声明；继承的身份字段一并列出。嵌套类型与完整数据字段见[类型声明](/reference/js/types)，Go / Java 以各自真实 DTO 为准。

```ts
interface Context {
    signal?: AbortSignal;
    timeout?: number;
    traceId?: string;
}

interface ConversationRef extends RuntimeRef {
    conversationId: string;
}

interface RuntimeRef {
    runtimeId: string;
}

interface Snapshot<T> {
    data: T;
    projection: Projection;
}

type Interaction = ConversationRef & {
    /** Store identity for stable presentation binding; interactionId remains the response ID. */
    canonicalInteractionId?: string;
    /** Equivalent response IDs, authoritative only within this exact owner/Turn/kind. */
    identityAliases?: string[];
    interactionId: string;
    turnId: string;
    state: 'pending' | 'resolving' | 'resolved' | 'expired' | 'orphaned' | 'unknown';
    outcome?: InteractionOutcome;
    deliveryState: 'dispatching' | 'provider_accepted' | 'delivered' | 'outcome_unknown' | 'unknown';
    title: string;
    revision: Int64;
    writeVersion?: string;
    agentId?: string;
    ownerConversationId?: string;
    expiresAt?: Int64;
    /** An explicit malformed deadline is never treated as unlimited validity. */
    validity?: 'unknown';
    unavailableReason?: 'IDENTITY_CONFLICT' | 'MISSING_AGENT' | 'EXECUTION_UNAVAILABLE' | 'DETAIL_NOT_SYNCHRONIZED' | 'AMBIGUOUS_REQUEST';
} & ({
    kind: 'permission';
    options: {
        optionId: string;
        label: string;
        kind?: string;
    }[];
} | {
    kind: 'question';
    questions: Question[];
    allowSkip: boolean;
});
```

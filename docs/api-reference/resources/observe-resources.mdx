## TypeScript 参数与结果结构

以下结构直接摘自同版本声明；继承的身份字段一并列出。嵌套类型与完整数据字段见[类型声明](/reference/js/types)，Go / Java 以各自真实 DTO 为准。

```ts
interface Context {
    signal?: AbortSignal;
    timeout?: number;
    traceId?: string;
}

interface RuntimeRef {
    runtimeId: string;
}

interface Resource {
    resourceId: string;
    runtimeId: string;
    kind: 'skill' | 'mcp' | 'attachment';
    name: string;
    availability: 'available' | 'unavailable' | 'unknown';
    revision: Int64;
    capabilities: {
        attach: Capability;
        preview: Capability;
    };
}

interface Observer<T> {
    onSnapshot(value: Snapshot<T>): void;
    onError(error: BusinessError$2): void;
}

interface Subscription {
    unsubscribe(): void;
}
```

`listConversations` · Conversation

返回 Page&lt;Conversation&gt;，读取 items；保留投影与公开分页信息。

## TypeScript 参数与结果结构

以下结构直接摘自同版本声明；继承的身份字段一并列出。嵌套类型与完整数据字段见[类型声明](/reference/js/types)，Go / Java 以各自真实 DTO 为准。

```ts
interface Context {
    signal?: AbortSignal;
    timeout?: number;
    traceId?: string;
}

interface ConversationQuery {
    /** Omit to query all cached Runtime sources of the authenticated account. */
    runtimeId?: string;
    search?: string;
    archive?: 'active' | 'archived' | 'all';
    pin?: 'all' | 'pinned' | 'unpinned';
    order?: 'activity' | 'pinned';
    agentId?: string;
    limit?: number;
    cursor?: string;
}

interface ListOptions {
    limit?: number;
    cursor?: string;
}

interface Page<T> {
    items: T[];
    nextCursor: string | null;
    projection?: Projection;
}

interface Conversation extends ConversationRef {
    name: string;
    mode: 'single' | 'group';
    agentId?: string;
    revision: Int64;
    executionState: 'idle' | 'running' | 'error' | 'unknown';
    availability: 'available' | 'unavailable' | 'unknown';
    archived: boolean;
    pinned: boolean;
    lastActivityAt: Int64;
    header?: ConversationHeaderMetadata;
    presentation?: ConversationPresentation;
    capabilities: {
        send: Capability;
        configure: Capability;
        members: Capability;
        resources: Capability;
    };
}

interface ConversationRef extends RuntimeRef {
    conversationId: string;
}

interface RuntimeRef {
    runtimeId: string;
}

interface Projection {
    /** Scoped mode only: authorized control/protocol availability, independent of catalogs/history. */
    controlReady?: boolean;
    /** Present only with explicit scoped readiness; coverage refers to this domain. */
    readinessDomain?: 'agents' | 'models' | 'conversation' | 'summaries';
    phase: 'initializing' | 'ready' | 'stale' | 'resyncing' | 'unavailable';
    coverage: 'partial' | 'complete';
    generation: string;
    commitId?: string;
    viewVersion: string;
    statusVersion: string;
    lastSyncedAt?: Int64;
}
```

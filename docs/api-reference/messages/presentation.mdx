`getMessagePresentation` · Message

按展示覆盖确认内容完整；Turn 已结束不自动代表最终答案已完整同步。

以下结构直接摘自同版本声明；继承的身份字段一并列出。嵌套类型与完整数据字段见[类型声明](/reference/js/types)，Go / Java 以各自真实 DTO 为准。

```ts
interface Context {
    signal?: AbortSignal;
    timeout?: number;
    traceId?: string;
}

interface PresentationQuery extends ConversationRef {
    messageId: string;
    cursor?: string;
    limit?: number;
    consistency?: 'cache' | 'refresh';
}

interface ConversationRef extends RuntimeRef {
    conversationId: string;
}

interface RuntimeRef {
    runtimeId: string;
}

interface Snapshot<T> {
    data: T;
    projection: Projection;
}

interface MessagePresentation {
    schemaVersion: '1';
    message: Message;
    snapshotRevision: string;
    answer: {
        state: 'final' | 'partial' | 'none' | 'unknown';
        itemIds: string[];
        blocks: Block[];
        coverage: Coverage;
    };
    process: {
        items: PresentationItem[];
        nextCursor: string | null;
        coverage: Coverage;
    };
    timing?: ExecutionTiming;
}
```

```ts
interface ExecutionContext {
    signal?: AbortSignal;
    traceId?: string;
    timeout?: number;
}

interface GetModelConfigRequest {
    /** 必填：Agent 实例 ID */

    /** Agent 类型，默认 "OpenClaw" */
    agentProvider?: string;

    agentPlatform?: string;

type CallOption = (options: CallOptions) => void;

interface ModelConfig {
    modelTemplateId: string;

    defaultModel: string;
    channelSessionModel: string;

}
```

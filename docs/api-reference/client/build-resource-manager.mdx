## 语言与授权

| 语言 | 公开入口 | 状态 |
|---|---|---|
| JavaScript / TypeScript | `Client.buildResourceManager` | 已公开声明，需核对环境能力 |
| Go | `github.com/aliyun/wuying-sdk-go.Client.BuildResourceManager` | 已公开声明，需核对环境能力 |

## TypeScript 参数与结果结构

以下结构直接摘自同版本声明；继承的身份字段一并列出。嵌套类型与完整数据字段见[类型声明](/reference/js/types)，Go / Java 以各自真实 DTO 为准。

```ts
interface Context {
    signal?: AbortSignal;
    timeout?: number;
    traceId?: string;
}

interface ResourceManager {
    attachImGroupFile(ctx: Context, req: IMGroupFileRequest): Promise<IMGroupAttachmentResult>;
    getImGroupAttachmentOperation(ctx: Context, req: IMGroupFileRequest): Promise<IMGroupAttachmentResult>;
    stageDraftAttachment(ctx: Context, req: DraftRef & {
        file: Blob;
        name: string;
        uploadId: string;
    }): Promise<StagedAttachment>;
    listDraftAttachments: ResourceManager$1['listDraftAttachments'];
    commitDraftAttachments(ctx: Context, req: DraftRef & {
        conversationId: string;
        attachmentIds: string[];
    }): Promise<CommittedAttachment[]>;
    discardDraftAttachments(ctx: Context, req: DraftRef & {
        attachmentIds: string[];
    }): Promise<{
        state: 'discarded';
    }>;
    getDraftAttachmentPreview(ctx: Context, req: DraftRef & {
        attachmentId: string;
    }): Promise<Snapshot<AttachmentPreview>>;
    listWorkspaces: ResourceManager$1['listWorkspaces'];
    createWorkspace(ctx: Context, req: RuntimeRef & {
        name: string;
    }): Promise<Workspace>;
    listResources: ResourceManager$1['listResources'];
    observeResources(ctx: Context, req: RuntimeRef & {
        kind?: Resource['kind'];
    }, observer: Observer<Resource[]>): Promise<Subscription>;
    observeConversationResources(ctx: Context, req: ConversationRef, observer: Observer<ResourceSelection>): Promise<Subscription>;
    getConversationResources(ctx: Context, req: ConversationRef): Promise<Snapshot<ResourceSelection>>;
    setConversationResources(ctx: Context, req: ConversationRef & {
        resourceIds: string[];
        expectedVersion: string;
    }): Promise<ResourceSelection>;
    stageAttachment(ctx: Context, req: ConversationRef & {
        file: Blob;
        name: string;
    }): Promise<StagedAttachment>;
    /** Discard a not-yet-committed upload. Does not delete conversation history or cloud files. */
    discardAttachment(ctx: Context, req: ConversationRef & {
        attachmentId: string;
    }): Promise<{
        state: 'discarded';
    }>;
    getAttachmentAccess(ctx: Context, req: ConversationRef & {
        attachmentId: string;
        purpose: 'preview' | 'download';
    }): Promise<{
        url: string;
        expiresAt: Int64;
    }>;
}
```

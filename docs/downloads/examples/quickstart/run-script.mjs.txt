export async function runScript(runtimeManager, request, ctx = {}) {
  let resolveDone, rejectDone;
  const done = new Promise((resolve, reject) => {
    resolveDone = resolve;
    rejectDone = reject;
  });
  // A callback can arrive before runCommand returns its subscription.
  void done.catch(() => {});
  const subscription = await runtimeManager.runCommand({ ...ctx, timeout: ctx.timeout ?? 600_000 }, request, {
    onEvent(event) {
      if (event.kind === 'accepted') {
        console.log('Command receipt:', event.receipt);
      }
      if (event.kind === 'completed') {
        if (event.allSucceeded) resolveDone(event.results);
        else rejectDone(Object.assign(new Error('Remote script failed'), {
          receipt: event.receipt, results: event.results,
        }));
      }
    },
    onError: rejectDone,
  });
  try {
    return await done;
  } finally {
    subscription.unsubscribe();
  }
}

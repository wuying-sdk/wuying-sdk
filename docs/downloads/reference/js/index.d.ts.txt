/**
 * Internal type surface of the Agent IM message transport (agmsg/v1).
 *
 * Web only: this module must stay free of node builtins, the legacy OpenIM
 * client and the package root entry. Field names and closed value sets follow
 * the frozen wire contract `test/agmsg/protocol/CONTRACT.md`.
 */
/** Any legal JSON value. `false`, `0`, `""` and `null` are real values, never "missing". */
type JSONValue = null | boolean | number | string | JSONValue[] | {
    [key: string]: JSONValue;
};
/** Phase of one stream packet. `seq` 1 must be `start`; `end` closes the stream. */
type StreamPhase = 'start' | 'delta' | 'end';
/** One opaque stream packet as delivered to the app. `data` is never parsed. */
interface StreamDeliveryPacket {
    groupId: string;
    senderUserId: string;
    streamId: string;
    seq: string;
    phase?: StreamPhase;
    issuedAtMs?: number;
    messageId: string;
    clientMessageId: string;
    data: JSONValue;
    /** True when the packet was served from a `stream_history` replay, not a live delivery. */
    replayed: boolean;
}
/** closed 表示接收端已补齐已知结束水位；truncated 明确表示 minSeq 之前的数据已丢失。 */
type StreamLifecycleStatus = 'closed' | 'truncated' | 'not_found';
type StreamEvent = {
    type: 'packet';
    packet: StreamDeliveryPacket;
} | {
    type: 'lifecycle';
    groupId: string;
    senderUserId: string;
    streamId: string;
    status: StreamLifecycleStatus;
};

 * 异步回调函数类型

 *
 * @param callback 异步回调函数。操作完成后在异步上下文中调用
 * @returns 配置了异步回调的选项函数
 *
 * @example
 * ```typescript
 * await manager.sendTextMessage(context, req, {
 *   async: (result, err) => {
 *     if (err) {
 *       console.error('发送失败:', err);
 *       return;
 *     }
 *     const msg = result as Message;
 *     console.log('发送成功，消息ID:', msg.id);
 *   }
 * });
 * ```
 */

 *
 * @param opts 调用选项列表
 * @returns 提取出的异步回调函数；如果没有配置异步选项，返回 null
 *
 * 内部使用：在 Manager 方法内部从 opts 中提取 asyncCallback
 */
declare function CallOptionsFrom(opts: CallOption[]): AsyncCallback | null;
/**
 * 执行异步或同步调用
 *
 * @param callback 异步回调（如果为 null 则同步执行）
 * @param fn 实际执行的函数
 *
 * @example
 * ```typescript
 * // 在方法内部使用
 * const callback = CallOptionsFrom(opts);
 *
 * return executeWithCallback(callback, async () => {
 *   // 实际工作
 *   return await this.agentService.someMethod();
 * });
 * ```
 */

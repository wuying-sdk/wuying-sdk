	WaitForAllGroups bool // true 等待全部群完整补齐后 Ready。
}

	// SalesEndpoint is the endpoint for Sales/Credit APIs.
	// Default: "https://ecd.cn-shanghai.aliyuncs.com".

func WithAsync(callback func(result interface{}, err error)) CallOption
    WithAsync enables async mode. The callback is invoked in a goroutine when
    the operation completes with either the result or an error.

    Usage:

        // Synchronous (default)
        msg, err := mgr.SendTextMessage(ctx, req)

        // Asynchronous
        mgr.SendTextMessage(ctx, req, manager.WithAsync(func(result interface{}, err error) {
            msg := result.(*model.Message)
            // handle result or error
        }))

```java Java
ConversationManager manager = client.buildConversationManager(ctx);
CreateRequest create = new CreateRequest();
create.runtimeId = runtimeId;
create.agentId = agentId;
create.mode = "single";
create.name = "我的会话";

CompletionStage<MessageCompletion> result = manager
    .createConversation(create, ctx)
    .thenCompose(created -> {
        SendRequest send = new SendRequest();
        send.runtimeId = runtimeId;
        send.conversationId = created.conversationId;
        send.text = "你好，请介绍一下你能做什么。";
        send.clientCommandId = commandId;
        return manager.sendMessageAndWait(send, ctx);
    });
```

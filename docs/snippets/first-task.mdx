移动端 onCompleted 是宿主 SDKEvent 回调：先处理 error，仅成功终态且 data.turn.state 为 completed 时进入答案读取；onError 为宿主错误回调。数据文件、工作线程及稳定请求标识按 SDK Client 与快速开始准备。

```java Android
JSONObject create = new JSONObject().put("runtimeId", runtimeId)
    .put("agentId", agentId).put("mode", "single").put("name", "第一次对话");
client.request(Methods.conversationCreateConversation, create, event -> {
    if (event.error != null) { onError.accept(event.error); return; }
    if (!event.terminal) return;
    try {
        JSONObject conversation = (JSONObject) event.data;
        JSONObject send = new JSONObject().put("runtimeId", runtimeId)
            .put("conversationId", conversation.getString("conversationId"))
            .put("text", "你好，请只回复一句问候。")
            .put("clientCommandId", clientCommandId);
        client.request(Methods.conversationSendMessageAndWait,
            send, null, null, 600000, onCompleted);
    } catch (Exception error) { onError.accept(error); }
});
```

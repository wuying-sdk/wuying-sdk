client, err := wuying.NewClient(
	wuying.ClientOptions{AuthCode: authCode},
	&config.UserConfig{ClientID: clientID, DataPath: dataFile},

	return err
}

);

        .put("authCode", authCode)
        .put("clientId", clientId)
        .put("dataPath", dataFile);
Client client = AndroidClient.connect(config);
